(function(Scratch) {
    'use strict';

    class MultiplayerExtension {
        getInfo() {
            return {
                id: 'multiplayerclient',
                name: 'Multiplayer Client',
                color1: '#FF4C4C',
                blocks: [
                    {
                        opcode: 'connectToServer',
                        blockType: Scratch.BlockType.COMMAND,
                        text: 'Connect to server at [URL]',
                        arguments: {
                            URL: {
                                type: Scratch.ArgumentType.STRING,
                                defaultValue: 'ws://localhost:8080'
                            }
                        }
                    },
                    {
                        opcode: 'sendData',
                        blockType: Scratch.BlockType.COMMAND,
                        text: 'Send data: [DATA]',
                        arguments: {
                            DATA: {
                                type: Scratch.ArgumentType.STRING,
                                defaultValue: '{"x": 0, "y": 0}'
                            }
                        }
                    },
                    {
                        opcode: 'getLatestData',
                        blockType: Scratch.BlockType.REPORTER,
                        text: 'Latest received data',
                    },
                    {
                        opcode: 'isConnected',
                        blockType: Scratch.BlockType.BOOLEAN,
                        text: 'Is connected?',
                    },
                    {
                        opcode: 'closeConnection',
                        blockType: Scratch.BlockType.COMMAND,
                        text: 'Disconnect',
                    }
                ]
            };
        }

        _socket = null;
        _latestData = '';
        _isConnected = false;

        connectToServer(args) {
            const url = args.URL;
            if (this._socket && this._socket.readyState === WebSocket.OPEN) return;

            this._socket = new WebSocket(url);

            this._socket.onopen = () => {
                this._isConnected = true;
                console.log('Connected to multiplayer server.');
            };

            this._socket.onmessage = (event) => {
                // When receiving data, store it for the 'getLatestData' block
                this._latestData = event.data;
            };

            this._socket.onclose = () => {
                this._isConnected = false;
                console.log('Disconnected from server.');
            };

            this._socket.onerror = (error) => {
                console.error('WebSocket error:', error);
                this._isConnected = false;
            };
        }

        sendData(args) {
            if (this._socket && this._socket.readyState === WebSocket.OPEN) {
                this._socket.send(args.DATA);
            }
        }

        getLatestData() {
            return this._latestData;
        }

        isConnected() {
            return this._isConnected;
        }
        
        closeConnection() {
            if (this._socket && this._socket.readyState === WebSocket.OPEN) {
                this._socket.close();
            }
        }
    }

    Scratch.extensions.register(new MultiplayerExtension());
})(Scratch);
